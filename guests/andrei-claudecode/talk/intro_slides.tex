\begin{frame}{How often does $p$ divide the number of trees?}{THE PROBLEM}
\begin{canvas}
  % the statement
  \at[base]{4.55,3.95}{\fs{15}$\vp\bigl(\Cq\bigr)\;=\;\dfrac{\sd(m)-1}{p-1}$}

  % what the letters mean
  \node[anchor=west,inner sep=0pt] at (8.35,3.8) {\fontsize{9}{15.5}\selectfont
    \begin{tabular}{@{}r@{\hspace{0.45cm}}l@{}}
      $q$      & a power of the prime $p$\\
      $m$      & the number of leaves, $(q-1)\,n+1$\\
      $\sd(m)$ & the sum of the digits of $m$ in base $p$\\
      $\vp$    & how many times $p$ divides
    \end{tabular}};

  % three examples
  \node[anchor=north,inner sep=0pt] at (8,5.2) {\fontsize{9.5}{12}\selectfont\renewcommand{\arraystretch}{1.45}
    \begin{tabular}{@{}l@{\hspace{1.0cm}}l@{\hspace{1.0cm}}c@{\hspace{0.9cm}}c@{\hspace{0.9cm}}c@{}}
      & \quiet{trees} & \quiet{leaves} & \quiet{in base $p$} & \quiet{power of $p$}\\\hline
      binary, $n=3$  & $5$ & $4$ & $100_2$ & \hl{0}\\
      ternary, $n=2$ & $3$ & $5$ & $12_3$ & \hl{1}\\
      ternary, $n=8$ & $43263=3^2\cdot 4807$ & $17$ & $122_3$ & \hl{2}
    \end{tabular}};
\end{canvas}
\end{frame}
